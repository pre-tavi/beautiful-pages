// The universe is not random. It is recursive beauty.

type Particle = {
  position: Vec3;
  velocity: Vec3;
  mass: number;
};

const attract = (a: Particle, b: Particle): Force => {
  const r = distance(a.position, b.position);
  const strength = (a.mass * b.mass) / (r * r);
  return direction(a, b).scale(strength);
};

// From two bodies, galaxies are born.
const universe = (particles: Particle[]): Universe =>
  particles.map(p =>
    integrate(p, particles.filter(other => other !== p)
      .map(other => attract(p, other))
    )
  );
